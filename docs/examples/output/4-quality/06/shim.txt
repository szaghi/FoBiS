subroutine gauss_legendre_gauss_nodes(n, x, w) bind(c, name="gauss_legendre_gauss_nodes")

  integer :: n_f
  real(c_double), intent(out) :: x(n)

  real(c_double), intent(out) :: w(n)

end subroutine gauss_legendre_gauss_nodes
